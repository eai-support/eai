These examples show how to call the EAI Platform API directly from the command line using `curl` and `jq`. They work in any Bash-compatible shell.

<Aside>
  All examples assume you have `BASE_URL_PUBLIC_API` set in your environment and a valid access token. Run `eai env pull` to populate your `.env.local` and `eai login` to authenticate.
</Aside>

## Setup

Export your environment variables before running any commands:

```bash
export BASE_URL="${BASE_URL_PUBLIC_API}"
export TENANT_ID="your-tenant-id"
export TOKEN="your-access-token"
```

You can source these from your `.env.local` file:

```bash
# Load variables from .env.local
set -a
source .env.local
set +a

export BASE_URL="${BASE_URL_PUBLIC_API}"
export TOKEN="${EAI_ACCESS_TOKEN}"
```

Define a reusable helper function for authenticated requests:

```bash
eai_request() {
  local method="$1"
  local path="$2"
  local data="$3"

  if [ -n "$data" ]; then
    curl -s -X "$method" \
      "${BASE_URL}${path}" \
      -H "Authorization: Bearer ${TOKEN}" \
      -H "Content-Type: application/json" \
      -d "$data"
  else
    curl -s -X "$method" \
      "${BASE_URL}${path}" \
      -H "Authorization: Bearer ${TOKEN}" \
      -H "Content-Type: application/json"
  fi
}
```

## Authentication

Verify your token and retrieve your user profile:

```bash
# Get the authenticated user profile
curl -s "${BASE_URL}/v3/auth/me" \
  -H "Authorization: Bearer ${TOKEN}" | jq .

# Extract specific fields
curl -s "${BASE_URL}/v3/auth/me" \
  -H "Authorization: Bearer ${TOKEN}" \
  | jq '{name: .displayName, email: .email, roles: .roles}'

# Provision the current user
curl -s -X POST "${BASE_URL}/v3/users/provisionme" \
  -H "Authorization: Bearer ${TOKEN}" \
  -H "Content-Type: application/json"
```

One-liner to verify you are logged in:

```bash
curl -sf "${BASE_URL}/v3/auth/me" \
  -H "Authorization: Bearer ${TOKEN}" \
  | jq -r '"Logged in as \(.displayName) (\(.email))"'
```

## Resource CRUD

Create, list, get, update, and delete resources using curl:

### Create a resource

```bash
curl -s -X POST "${BASE_URL}/v3/resources/${TENANT_ID}/Task" \
  -H "Authorization: Bearer ${TOKEN}" \
  -H "Content-Type: application/json" \
  -d '{
    "title": "Review pull request",
    "status": "open",
    "priority": "high"
  }' | jq .
```

Store the created resource ID in a variable for subsequent operations:

```bash
TASK_ID=$(curl -s -X POST "${BASE_URL}/v3/resources/${TENANT_ID}/Task" \
  -H "Authorization: Bearer ${TOKEN}" \
  -H "Content-Type: application/json" \
  -d '{"title": "Review pull request", "status": "open", "priority": "high"}' \
  | jq -r '.id')

echo "Created task: ${TASK_ID}"
```

### List resources

```bash
curl -s "${BASE_URL}/v3/resources/${TENANT_ID}/Task" \
  -H "Authorization: Bearer ${TOKEN}" \
  | jq '.items[] | {id: .id, title: .data.title, status: .data.status}'
```

### Get a single resource

```bash
curl -s "${BASE_URL}/v3/resources/${TENANT_ID}/Task/${TASK_ID}" \
  -H "Authorization: Bearer ${TOKEN}" | jq .
```

### Update a resource

```bash
curl -s -X PUT "${BASE_URL}/v3/resources/${TENANT_ID}/Task/${TASK_ID}" \
  -H "Authorization: Bearer ${TOKEN}" \
  -H "Content-Type: application/json" \
  -d '{
    "title": "Review pull request",
    "status": "in-progress",
    "priority": "high"
  }' | jq .
```

### Delete a resource

```bash
curl -s -X DELETE "${BASE_URL}/v3/resources/${TENANT_ID}/Task/${TASK_ID}" \
  -H "Authorization: Bearer ${TOKEN}"

echo "Task ${TASK_ID} deleted"
```

### Full CRUD script

```bash
#!/usr/bin/env bash
set -euo pipefail

# Create
echo "Creating task..."
TASK_ID=$(eai_request POST "/v3/resources/${TENANT_ID}/Task" \
  '{"title": "Review pull request", "status": "open", "priority": "high"}' \
  | jq -r '.id')
echo "Created: ${TASK_ID}"

# List
echo "Listing tasks..."
TOTAL=$(eai_request GET "/v3/resources/${TENANT_ID}/Task" | jq '.total')
echo "Found ${TOTAL} tasks"

# Get
echo "Fetching task..."
TITLE=$(eai_request GET "/v3/resources/${TENANT_ID}/Task/${TASK_ID}" \
  | jq -r '.data.title')
echo "Title: ${TITLE}"

# Update
echo "Updating task..."
eai_request PUT "/v3/resources/${TENANT_ID}/Task/${TASK_ID}" \
  '{"title": "Review pull request", "status": "in-progress", "priority": "high"}' \
  | jq '.data.status'

# Delete
echo "Deleting task..."
eai_request DELETE "/v3/resources/${TENANT_ID}/Task/${TASK_ID}"
echo "Done"
```

## Querying resources

Fetch the tenant schema and inspect available object types:

```bash
# List all object type names
curl -s "${BASE_URL}/v3/resources/schema/${TENANT_ID}" \
  -H "Authorization: Bearer ${TOKEN}" \
  | jq -r '.objectTypes[].name'

# Get full schema details for a specific type
curl -s "${BASE_URL}/v3/resources/schema/${TENANT_ID}" \
  -H "Authorization: Bearer ${TOKEN}" \
  | jq '.objectTypes[] | select(.name == "Task")'

# Count properties per object type
curl -s "${BASE_URL}/v3/resources/schema/${TENANT_ID}" \
  -H "Authorization: Bearer ${TOKEN}" \
  | jq '.objectTypes[] | {name: .name, propertyCount: (.properties | length)}'
```

#!/usr/bin/env bash
set -euo pipefail

WORKFLOW_ID="wf-123"
STAGE="default"

echo "EAI Chat (type 'exit' to quit)"
echo "---"

while true; do
  printf "> "
  read -r user_input

  if [ "$user_input" = "exit" ]; then
    echo "Goodbye."
    break
  fi

  PAYLOAD=$(jq -n --arg msg "$user_input" '{"message": $msg}')

  REPLY=$(curl -s -X POST \
    "${BASE_URL}/v3/chat/${TENANT_ID}/${WORKFLOW_ID}/${STAGE}" \
    -H "Authorization: Bearer ${TOKEN}" \
    -H "Content-Type: application/json" \
    -d "$PAYLOAD" \
    | jq -r '.message')

  echo "AI: ${REPLY}"
  echo ""
done

## Error handling

Check HTTP status codes and handle errors in shell scripts:

```bash
#!/usr/bin/env bash
set -euo pipefail

eai_safe_request() {
  local method="$1"
  local path="$2"
  local data="${3:-}"

  local http_code body
  if [ -n "$data" ]; then
    response=$(curl -s -w "\n%{http_code}" -X "$method" \
      "${BASE_URL}${path}" \
      -H "Authorization: Bearer ${TOKEN}" \
      -H "Content-Type: application/json" \
      -d "$data")
  else
    response=$(curl -s -w "\n%{http_code}" -X "$method" \
      "${BASE_URL}${path}" \
      -H "Authorization: Bearer ${TOKEN}" \
      -H "Content-Type: application/json")
  fi

  http_code=$(echo "$response" | tail -n1)
  body=$(echo "$response" | sed '$d')

  case "$http_code" in
    2*)
      echo "$body"
      ;;
    401)
      echo "Error: Token expired. Run 'eai login' to re-authenticate." >&2
      return 1
      ;;
    404)
      echo "Error: Resource not found." >&2
      return 1
      ;;
    429)
      echo "Warning: Rate limited. Retrying in 2 seconds..." >&2
      sleep 2
      eai_safe_request "$method" "$path" "$data"
      ;;
    *)
      echo "Error: API returned HTTP ${http_code}" >&2
      echo "$body" | jq . >&2 2>/dev/null || echo "$body" >&2
      return 1
      ;;
  esac
}

# Usage
eai_safe_request GET "/v3/resources/${TENANT_ID}/Task" | jq '.total'
eai_safe_request GET "/v3/resources/${TENANT_ID}/Task/nonexistent-id"
```
