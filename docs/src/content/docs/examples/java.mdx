These examples show how to call the EAI Platform API from Java 11+ using the built-in `java.net.http.HttpClient` and Jackson for JSON serialization.

<Aside>
  All examples assume you have `BASE_URL_PUBLIC_API` and your access token available as environment variables. Run `eai env pull` to populate your `.env.local` file.
</Aside>

## Setup

```java
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.time.Duration;

public class EaiClient {

    private final HttpClient httpClient;
    private final ObjectMapper mapper;
    private final String baseUrl;
    private final String tenantId;
    private final String accessToken;

    public EaiClient(String baseUrl, String tenantId, String accessToken) {
        this.baseUrl = baseUrl;
        this.tenantId = tenantId;
        this.accessToken = accessToken;
        this.mapper = new ObjectMapper();
        this.httpClient = HttpClient.newBuilder()
            .connectTimeout(Duration.ofSeconds(10))
            .build();
    }

    private JsonNode sendRequest(String method, String path, Object body) throws Exception {
        HttpRequest.Builder builder = HttpRequest.newBuilder()
            .uri(URI.create(baseUrl + path))
            .header("Authorization", "Bearer " + accessToken)
            .header("Content-Type", "application/json")
            .timeout(Duration.ofSeconds(30));

        if (body != null) {
            String json = mapper.writeValueAsString(body);
            builder.method(method, HttpRequest.BodyPublishers.ofString(json));
        } else {
            builder.method(method, HttpRequest.BodyPublishers.noBody());
        }

        HttpResponse<String> response = httpClient.send(
            builder.build(), HttpResponse.BodyHandlers.ofString());

        if (response.statusCode() >= 400) {
            throw new EaiApiException(response.statusCode(), response.body());
        }

        return response.body().isEmpty() ? null : mapper.readTree(response.body());
    }
}
```

## Authentication

Verify your token and retrieve your user profile:

```java
public JsonNode getAuthenticatedUser() throws Exception {
    return sendRequest("GET", "/v3/auth/me", null);
}

public void provisionUser() throws Exception {
    sendRequest("POST", "/v3/users/provisionme", null);
}

// Usage
public static void main(String[] args) throws Exception {
    String baseUrl = System.getenv("BASE_URL_PUBLIC_API");
    String tenantId = System.getenv("TENANT_ID");
    String token = System.getenv("EAI_ACCESS_TOKEN");

    EaiClient client = new EaiClient(baseUrl, tenantId, token);

    JsonNode me = client.getAuthenticatedUser();
    System.out.printf("Logged in as %s (%s)%n",
        me.get("displayName").asText(),
        me.get("email").asText());
}
```

## Resource CRUD

Create, list, get, update, and delete resources:

```java
public JsonNode createResource(String objectType, Object data) throws Exception {
    return sendRequest("POST",
        "/v3/resources/" + tenantId + "/" + objectType, data);
}

public JsonNode listResources(String objectType) throws Exception {
    return sendRequest("GET",
        "/v3/resources/" + tenantId + "/" + objectType, null);
}

public JsonNode getResource(String objectType, String resourceId) throws Exception {
    return sendRequest("GET",
        "/v3/resources/" + tenantId + "/" + objectType + "/" + resourceId, null);
}

public JsonNode updateResource(String objectType, String resourceId, Object data)
        throws Exception {
    return sendRequest("PUT",
        "/v3/resources/" + tenantId + "/" + objectType + "/" + resourceId, data);
}

public void deleteResource(String objectType, String resourceId) throws Exception {
    sendRequest("DELETE",
        "/v3/resources/" + tenantId + "/" + objectType + "/" + resourceId, null);
}
```

### Full CRUD workflow

```java
import java.util.Map;

// Create
JsonNode task = client.createResource("Task", Map.of(
    "title", "Review pull request",
    "status", "open",
    "priority", "high"
));
String taskId = task.get("id").asText();
System.out.println("Created task " + taskId);

// List
JsonNode tasks = client.listResources("Task");
System.out.printf("Found %d tasks%n", tasks.get("total").asInt());

// Get
JsonNode fetched = client.getResource("Task", taskId);
System.out.println("Task title: " + fetched.get("data").get("title").asText());

// Update
JsonNode updated = client.updateResource("Task", taskId, Map.of(
    "title", "Review pull request",
    "status", "in-progress",
    "priority", "high"
));
System.out.println("Updated status to " + updated.get("data").get("status").asText());

// Delete
client.deleteResource("Task", taskId);
System.out.println("Task deleted");
```

```java
import java.net.http.HttpResponse.BodyHandlers;
import java.io.InputStream;

public void streamChat(String workflowId, String stage, String message)
        throws Exception {
    String path = "/v3/chat/stream/" + tenantId + "/" + workflowId + "/" + stage;
    String json = mapper.writeValueAsString(Map.of("message", message));

    HttpRequest request = HttpRequest.newBuilder()
        .uri(URI.create(baseUrl + path))
        .header("Authorization", "Bearer " + accessToken)
        .header("Content-Type", "application/json")
        .POST(HttpRequest.BodyPublishers.ofString(json))
        .build();

    HttpResponse<InputStream> response = httpClient.send(
        request, BodyHandlers.ofInputStream());

    if (response.statusCode() >= 400) {
        throw new EaiApiException(response.statusCode(), "Stream request failed");
    }

    try (InputStream stream = response.body()) {
        byte[] buffer = new byte[256];
        int bytesRead;
        while ((bytesRead = stream.read(buffer)) != -1) {
            System.out.print(new String(buffer, 0, bytesRead));
        }
    }

    System.out.println();
}

client.streamChat("wf-123", "default", "What is the project status?");
```

```java
import java.util.concurrent.CompletableFuture;

public CompletableFuture<JsonNode> sendChatMessageAsync(
        String workflowId, String stage, String message) {
    String path = "/v3/chat/" + tenantId + "/" + workflowId + "/" + stage;
    String json;
    try {
        json = mapper.writeValueAsString(Map.of("message", message));
    } catch (Exception e) {
        return CompletableFuture.failedFuture(e);
    }

    HttpRequest request = HttpRequest.newBuilder()
        .uri(URI.create(baseUrl + path))
        .header("Authorization", "Bearer " + accessToken)
        .header("Content-Type", "application/json")
        .POST(HttpRequest.BodyPublishers.ofString(json))
        .build();

    return httpClient.sendAsync(request, HttpResponse.BodyHandlers.ofString())
        .thenApply(response -> {
            if (response.statusCode() >= 400) {
                throw new RuntimeException(
                    new EaiApiException(response.statusCode(), response.body()));
            }
            try {
                return mapper.readTree(response.body());
            } catch (Exception e) {
                throw new RuntimeException(e);
            }
        });
}

// Usage
client.sendChatMessageAsync("wf-123", "default", "Summarize open tasks")
    .thenAccept(reply -> System.out.println("AI: " + reply.get("message").asText()))
    .join();
```

## Error handling

Define a custom exception and handle common error codes:

```java
public class EaiApiException extends Exception {

    private final int statusCode;
    private final String responseBody;

    public EaiApiException(int statusCode, String responseBody) {
        super("EAI API error " + statusCode + ": " + responseBody);
        this.statusCode = statusCode;
        this.responseBody = responseBody;
    }

    public int getStatusCode() { return statusCode; }
    public String getResponseBody() { return responseBody; }
    public boolean isUnauthorized() { return statusCode == 401; }
    public boolean isNotFound() { return statusCode == 404; }
    public boolean isRateLimited() { return statusCode == 429; }
}

// Usage with structured error handling
try {
    JsonNode tasks = client.listResources("Task");
    System.out.printf("Found %d tasks%n", tasks.get("total").asInt());
} catch (EaiApiException ex) {
    if (ex.isUnauthorized()) {
        System.err.println("Token expired. Run `eai login` to re-authenticate.");
    } else if (ex.isRateLimited()) {
        System.err.println("Rate limited. Retrying after delay...");
        Thread.sleep(2000);
        JsonNode tasks = client.listResources("Task");
        System.out.printf("Found %d tasks%n", tasks.get("total").asInt());
    } else {
        System.err.println("API error: " + ex.getMessage());
    }
}
```
