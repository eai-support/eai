These examples show how to call the EAI Platform API from Go using the standard library. All requests use `context.Context` for cancellation and `encoding/json` for serialization.

<Aside>
  All examples assume you have `BASE_URL_PUBLIC_API` and your access token available as environment variables. Run `eai env pull` to populate your `.env.local` file.
</Aside>

## Setup

```go
package eai

import (
	"bytes"
	"context"
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"os"
	"time"
)

type Client struct {
	BaseURL    string
	TenantID   string
	Token      string
	HTTPClient *http.Client
}

func NewClient() *Client {
	return &Client{
		BaseURL:  os.Getenv("BASE_URL_PUBLIC_API"),
		TenantID: os.Getenv("TENANT_ID"),
		Token:    os.Getenv("EAI_ACCESS_TOKEN"),
		HTTPClient: &http.Client{
			Timeout: 30 * time.Second,
		},
	}
}

func (c *Client) doRequest(ctx context.Context, method, path string, body any) ([]byte, error) {
	var reqBody io.Reader
	if body != nil {
		data, err := json.Marshal(body)
		if err != nil {
			return nil, fmt.Errorf("marshal request body: %w", err)
		}
		reqBody = bytes.NewReader(data)
	}

	req, err := http.NewRequestWithContext(ctx, method, c.BaseURL+path, reqBody)
	if err != nil {
		return nil, fmt.Errorf("create request: %w", err)
	}

	req.Header.Set("Authorization", "Bearer "+c.Token)
	req.Header.Set("Content-Type", "application/json")

	resp, err := c.HTTPClient.Do(req)
	if err != nil {
		return nil, fmt.Errorf("send request: %w", err)
	}
	defer resp.Body.Close()

	respBody, err := io.ReadAll(resp.Body)
	if err != nil {
		return nil, fmt.Errorf("read response: %w", err)
	}

	if resp.StatusCode >= 400 {
		return nil, &ApiError{StatusCode: resp.StatusCode, Body: string(respBody)}
	}

	return respBody, nil
}
```

## Authentication

Verify your token and retrieve your user profile:

```go
type UserProfile struct {
	ID          string   `json:"id"`
	Email       string   `json:"email"`
	DisplayName string   `json:"displayName"`
	Roles       []string `json:"roles"`
}

func (c *Client) GetAuthenticatedUser(ctx context.Context) (*UserProfile, error) {
	data, err := c.doRequest(ctx, http.MethodGet, "/v3/auth/me", nil)
	if err != nil {
		return nil, err
	}
	var user UserProfile
	if err := json.Unmarshal(data, &user); err != nil {
		return nil, fmt.Errorf("decode user profile: %w", err)
	}
	return &user, nil
}

func (c *Client) ProvisionUser(ctx context.Context) error {
	_, err := c.doRequest(ctx, http.MethodPost, "/v3/users/provisionme", nil)
	return err
}

// Usage
func main() {
	client := NewClient()
	ctx := context.Background()

	user, err := client.GetAuthenticatedUser(ctx)
	if err != nil {
		log.Fatalf("auth failed: %v", err)
	}
	fmt.Printf("Logged in as %s (%s)\n", user.DisplayName, user.Email)
}
```

## Resource CRUD

Create, list, get, update, and delete resources:

```go
type Resource struct {
	ID         string         `json:"id"`
	ObjectType string         `json:"objectType"`
	Data       map[string]any `json:"data"`
	CreatedAt  time.Time      `json:"createdAt"`
	UpdatedAt  time.Time      `json:"updatedAt"`
}

type ResourceList struct {
	Items    []Resource `json:"items"`
	Total    int        `json:"total"`
	Page     int        `json:"page"`
	PageSize int        `json:"pageSize"`
}

func (c *Client) CreateResource(ctx context.Context, objectType string, data any) (*Resource, error) {
	path := fmt.Sprintf("/v3/resources/%s/%s", c.TenantID, objectType)
	respBody, err := c.doRequest(ctx, http.MethodPost, path, data)
	if err != nil {
		return nil, err
	}
	var resource Resource
	return &resource, json.Unmarshal(respBody, &resource)
}

func (c *Client) ListResources(ctx context.Context, objectType string) (*ResourceList, error) {
	path := fmt.Sprintf("/v3/resources/%s/%s", c.TenantID, objectType)
	respBody, err := c.doRequest(ctx, http.MethodGet, path, nil)
	if err != nil {
		return nil, err
	}
	var list ResourceList
	return &list, json.Unmarshal(respBody, &list)
}

func (c *Client) GetResource(ctx context.Context, objectType, resourceID string) (*Resource, error) {
	path := fmt.Sprintf("/v3/resources/%s/%s/%s", c.TenantID, objectType, resourceID)
	respBody, err := c.doRequest(ctx, http.MethodGet, path, nil)
	if err != nil {
		return nil, err
	}
	var resource Resource
	return &resource, json.Unmarshal(respBody, &resource)
}

func (c *Client) UpdateResource(ctx context.Context, objectType, resourceID string, data any) (*Resource, error) {
	path := fmt.Sprintf("/v3/resources/%s/%s/%s", c.TenantID, objectType, resourceID)
	respBody, err := c.doRequest(ctx, http.MethodPut, path, data)
	if err != nil {
		return nil, err
	}
	var resource Resource
	return &resource, json.Unmarshal(respBody, &resource)
}

func (c *Client) DeleteResource(ctx context.Context, objectType, resourceID string) error {
	path := fmt.Sprintf("/v3/resources/%s/%s/%s", c.TenantID, objectType, resourceID)
	_, err := c.doRequest(ctx, http.MethodDelete, path, nil)
	return err
}
```

### Full CRUD workflow

```go
ctx := context.Background()

// Create
task, err := client.CreateResource(ctx, "Task", map[string]any{
	"title":    "Review pull request",
	"status":   "open",
	"priority": "high",
})
if err != nil {
	log.Fatalf("create: %v", err)
}
fmt.Printf("Created task %s\n", task.ID)

// List
tasks, err := client.ListResources(ctx, "Task")
if err != nil {
	log.Fatalf("list: %v", err)
}
fmt.Printf("Found %d tasks\n", tasks.Total)

// Get
fetched, err := client.GetResource(ctx, "Task", task.ID)
if err != nil {
	log.Fatalf("get: %v", err)
}
fmt.Printf("Task title: %s\n", fetched.Data["title"])

// Update
updated, err := client.UpdateResource(ctx, "Task", task.ID, map[string]any{
	"title":    "Review pull request",
	"status":   "in-progress",
	"priority": "high",
})
if err != nil {
	log.Fatalf("update: %v", err)
}
fmt.Printf("Updated status to %s\n", updated.Data["status"])

// Delete
if err := client.DeleteResource(ctx, "Task", task.ID); err != nil {
	log.Fatalf("delete: %v", err)
}
fmt.Println("Task deleted")
```

type SchemaResponse struct {

func (c *Client) StreamChat(ctx context.Context, workflowID, stage, message string) error {
	path := fmt.Sprintf("/v3/chat/stream/%s/%s/%s", c.TenantID, workflowID, stage)
	body, _ := json.Marshal(map[string]string{"message": message})

	req, err := http.NewRequestWithContext(ctx, http.MethodPost, c.BaseURL+path, bytes.NewReader(body))
	if err != nil {
		return fmt.Errorf("create request: %w", err)
	}
	req.Header.Set("Authorization", "Bearer "+c.Token)
	req.Header.Set("Content-Type", "application/json")

	resp, err := c.HTTPClient.Do(req)
	if err != nil {
		return fmt.Errorf("send request: %w", err)
	}
	defer resp.Body.Close()

	if resp.StatusCode >= 400 {
		errBody, _ := io.ReadAll(resp.Body)
		return &ApiError{StatusCode: resp.StatusCode, Body: string(errBody)}
	}

	buf := make([]byte, 256)
	for {
		n, err := resp.Body.Read(buf)
		if n > 0 {
			fmt.Print(string(buf[:n]))
		}
		if err == io.EOF {
			break
		}
		if err != nil {
			return fmt.Errorf("read stream: %w", err)
		}
	}
	fmt.Println()
	return nil
}

## Error handling

Define a custom error type and handle common status codes:

```go
type ApiError struct {
	StatusCode int
	Body       string
}

func (e *ApiError) Error() string {
	return fmt.Sprintf("EAI API error %d: %s", e.StatusCode, e.Body)
}

func (e *ApiError) IsUnauthorized() bool { return e.StatusCode == 401 }
func (e *ApiError) IsNotFound() bool     { return e.StatusCode == 404 }
func (e *ApiError) IsRateLimited() bool  { return e.StatusCode == 429 }

// Usage with structured error handling
tasks, err := client.ListResources(ctx, "Task")
if err != nil {
	var apiErr *ApiError
	if errors.As(err, &apiErr) {
		switch {
		case apiErr.IsUnauthorized():
			fmt.Fprintln(os.Stderr, "Token expired. Run `eai login` to re-authenticate.")
		case apiErr.IsRateLimited():
			fmt.Fprintln(os.Stderr, "Rate limited. Retrying after delay...")
			time.Sleep(2 * time.Second)
			tasks, err = client.ListResources(ctx, "Task")
		default:
			fmt.Fprintf(os.Stderr, "API error: %v\n", apiErr)
		}
	} else {
		log.Fatalf("unexpected error: %v", err)
	}
}
if tasks != nil {
	fmt.Printf("Found %d tasks\n", tasks.Total)
}
```
