These examples show how to interact with the EAI Platform API from TypeScript using the native `fetch` API. Every request requires a Bearer token obtained through Entra CIAM browser-based authorization code flow with PKCE.

<Aside>
  All examples assume you have `BASE_URL_PUBLIC_API` set in your environment. Run `eai env pull` to populate your `.env.local` file.
</Aside>

## Setup

```typescript
const BASE_URL = process.env.BASE_URL_PUBLIC_API!;
const TENANT_ID = process.env.TENANT_ID!;
const TOKEN = process.env.EAI_ACCESS_TOKEN!;

interface RequestOptions {
  method?: string;
  body?: unknown;
}

async function eaiFetch<T>(path: string, options: RequestOptions = {}): Promise<T> {
  const response = await fetch(`${BASE_URL}${path}`, {
    method: options.method ?? "GET",
    headers: {
      Authorization: `Bearer ${TOKEN}`,
      "Content-Type": "application/json",
    },
    body: options.body ? JSON.stringify(options.body) : undefined,
  });

  if (!response.ok) {
    const errorBody = await response.text();
    throw new Error(`API error ${response.status}: ${errorBody}`);
  }

  return response.json() as Promise<T>;
}
```

## Authentication

Verify your token and retrieve your user profile:

```typescript
interface UserProfile {
  id: string;
  email: string;
  displayName: string;
  roles: string[];
}

async function getAuthenticatedUser(): Promise<UserProfile> {
  return eaiFetch<UserProfile>("/v3/auth/me");
}

async function provisionUser(): Promise<void> {
  await eaiFetch<void>("/v3/users/provisionme", { method: "POST" });
}

// Usage
const me = await getAuthenticatedUser();
console.log(`Logged in as ${me.displayName} (${me.email})`);
```

## Resource CRUD

Define types for your resources and perform create, read, update, list, and delete operations:

```typescript
interface Resource {
  id: string;
  objectType: string;
  data: Record<string, unknown>;
  createdAt: string;
  updatedAt: string;
}

interface ResourceList {
  items: Resource[];
  total: number;
  page: number;
  pageSize: number;
}

// Create a resource
async function createResource(objectType: string, data: Record<string, unknown>): Promise<Resource> {
  return eaiFetch<Resource>(`/v3/resources/${TENANT_ID}/${objectType}`, {
    method: "POST",
    body: data,
  });
}

// List resources of a given type
async function listResources(objectType: string): Promise<ResourceList> {
  return eaiFetch<ResourceList>(`/v3/resources/${TENANT_ID}/${objectType}`);
}

// Get a single resource by ID
async function getResource(objectType: string, resourceId: string): Promise<Resource> {
  return eaiFetch<Resource>(`/v3/resources/${TENANT_ID}/${objectType}/${resourceId}`);
}

// Update a resource
async function updateResource(
  objectType: string,
  resourceId: string,
  data: Record<string, unknown>,
): Promise<Resource> {
  return eaiFetch<Resource>(`/v3/resources/${TENANT_ID}/${objectType}/${resourceId}`, {
    method: "PUT",
    body: data,
  });
}

// Delete a resource
async function deleteResource(objectType: string, resourceId: string): Promise<void> {
  await eaiFetch<void>(`/v3/resources/${TENANT_ID}/${objectType}/${resourceId}`, {
    method: "DELETE",
  });
}
```

### Full CRUD workflow

```typescript
// Create
const task = await createResource("Task", {
  title: "Review pull request",
  status: "open",
  priority: "high",
});
console.log(`Created task ${task.id}`);

// List
const tasks = await listResources("Task");
console.log(`Found ${tasks.total} tasks`);

// Get
const fetched = await getResource("Task", task.id);
console.log(`Task title: ${fetched.data.title}`);

// Update
const updated = await updateResource("Task", task.id, {
  ...fetched.data,
  status: "in-progress",
});
console.log(`Updated status to ${updated.data.status}`);

// Delete
await deleteResource("Task", task.id);
console.log("Task deleted");
```

## Querying resources

Fetch the schema for a tenant and filter resources with query parameters:

interface SchemaResponse {

async function getSchema(): Promise<SchemaResponse> {

```typescript
async function streamChat(
  workflowId: string,
  stage: string,
  message: string,
): Promise<void> {
  const response = await fetch(
    `${BASE_URL}/v3/chat/stream/${TENANT_ID}/${workflowId}/${stage}`,
    {
      method: "POST",
      headers: {
        Authorization: `Bearer ${TOKEN}`,
        "Content-Type": "application/json",
      },
      body: JSON.stringify({ message }),
    },
  );

  if (!response.ok || !response.body) {
    throw new Error(`Stream error: ${response.status}`);
  }

  const reader = response.body.getReader();
  const decoder = new TextDecoder();

  while (true) {
    const { done, value } = await reader.read();
    if (done) break;
    process.stdout.write(decoder.decode(value, { stream: true }));
  }

  console.log(); // Final newline
}

await streamChat("wf-123", "default", "What is the project status?");
```

## Error handling

Wrap API calls with structured error handling:

```typescript
class EaiApiError extends Error {
  constructor(
    public readonly status: number,
    public readonly body: string,
  ) {
    super(`EAI API error ${status}: ${body}`);
    this.name = "EaiApiError";
  }

  get isUnauthorized(): boolean {
    return this.status === 401;
  }

  get isNotFound(): boolean {
    return this.status === 404;
  }

  get isRateLimited(): boolean {
    return this.status === 429;
  }
}

async function safeRequest<T>(fn: () => Promise<T>): Promise<T> {
  try {
    return await fn();
  } catch (error) {
    if (error instanceof EaiApiError) {
      if (error.isUnauthorized) {
        console.error("Token expired. Run `eai login` to re-authenticate.");
      } else if (error.isRateLimited) {
        console.error("Rate limited. Retrying after delay...");
        await new Promise((resolve) => setTimeout(resolve, 2000));
        return fn();
      } else {
        console.error(`API error: ${error.message}`);
      }
    }
    throw error;
  }
}

// Usage
const result = await safeRequest(() => listResources("Task"));
console.log(`Found ${result.total} tasks`);
```
